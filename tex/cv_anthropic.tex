\section{Selected Research and Scientific Software}
\denseouterlist{
\entrymid[\textbullet]
{\textbf{Bayesian noise modelling and validation} --- \link{https://github.com/nz-gravity/LogPSplinePSD}{LogPSplinePSD}}{}
{Develop P-spline spectral models and JAX/NumPyro inference for gravitational-wave
detector noise. First-author multivariate study uses simulations with known
spectra to test uncertainty coverage and shows when a diagonal-covariance
approximation fails for asymmetric LISA noise. Co-authored variational-inference
study compares spectral estimates and computational cost with HMC targeting
the same posterior.}

\entrymid[\textbullet]
{\textbf{Differentiable time--frequency analysis} --- \link{https://github.com/pywavelet/wdm_transform}{wdm-transform}}{}
{First author and software developer of an explicit Wilson--Daubechies--Meyer
transform. Connect mathematical derivations with NumPy/JAX implementations
and numerical checks of reconstruction, orthogonality and coefficient covariance.
A controlled LISA example verifies agreement between wavelet- and
frequency-domain posteriors under a shared stationary noise model.}

\entrymid[\textbullet]
{\textbf{Surrogates for expensive simulations} --- \link{https://github.com/COMPAS-Surrogate/lnl_surrogate}{COMPAS LnL Surrogate}}{}
{Built a Gaussian-process likelihood-surrogate package for COMPAS binary-star
population simulations. Implemented Bayesian optimisation and active learning
with exploratory and exploitative acquisition functions; included mock-likelihood
examples to check predictions and uncertainty before applying the workflow
to expensive simulations.}

\entrymid[\textbullet]
{\textbf{Generative waveform models} --- \link{https://github.com/starccato/starccato_jax}{Starccato-JAX}}{}
{Co-develop JAX variational autoencoders for core-collapse supernova and
detector-glitch waveforms, with pretrained decoders and reproducible latent-space
sampling. Provide day-to-day research guidance on related supernova-detection
studies using normalising flows.}
}
